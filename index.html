function buildInnings(withC,pres){
  var n=pres.length,spots=Math.min(withC?9:8,n),bpc=n-spots,r=[{},{},{}];if(n<2)return r;
  for(var i=0;i<3;i++){
    var st=(i*bpc)%n;for(var j=0;j<bpc;j++)r[i][pres[(st+j)%n]]='Bench';
    var pi=(i*3)%n,ptch=pres[pi];if(r[i][ptch])ptch=pres[(pi+1)%n];if(!r[i][ptch])r[i][ptch]='Pitcher';
    if(withC&&spots>=2){var ci=(i*3+2)%n,att=0,cat=pres[ci];while((r[i][cat]||cat===ptch)&&att<n){att++;cat=pres[(ci+att)%n];}if(!r[i][cat])r[i][cat]='Catcher';}
    var rem=pres.filter(function(p){return !r[i][p];}),pool=shuf(POS).slice(0,rem.length);
    for(var k=0;k<rem.length;k++)r[i][rem[k]]=pool[k]||'Left Field';
  }
  return r;
}

function build4(withC,pres,inn){
  var n=pres.length;if(n<2)return{inn4:{},bc:{}};
  var spots=Math.min(withC?9:8,n),bc={};
  pres.forEach(function(p){bc[p]=[0,1,2].filter(function(i){return inn[i][p]==='Bench';}).length;});
  var sorted=pres.slice().sort(function(a,b){return bc[b]-bc[a];}),benched=sorted.slice(spots),field=sorted.slice(0,spots),inn4={};
  benched.forEach(function(p){inn4[p]='Bench';});
  var ptch=field[0];inn4[ptch]='Pitcher';
  if(withC&&field.length>=2){var cat=field[1];if(!inn4[cat])inn4[cat]='Catcher';}
  var rem=field.filter(function(p){return !inn4[p];}),pool=shuf(POS).slice(0,rem.length);
  rem.forEach(function(p,j){inn4[p]=pool[j]||'Left Field';});
  return{inn4:inn4,bc:bc};
}

function hashPin(pin, cb){
  var msgBuffer = new TextEncoder().encode(pin);
  crypto.subtle.digest('SHA-256', msgBuffer).then(function(hashBuffer){
    var hashArray = Array.from(new Uint8Array(hashBuffer));
    var hashHex = hashArray.map(function(b){return b.toString(16).padStart(2,'0');}).join('');
    cb(hashHex);
  });
}

  var err=el('pin-error');
  if(!inp)return;
  var val=inp.value;
  if(!val){if(err)err.textContent='Please enter the passphrase.';return;}
  hashPin(val,function(h){
    if(h===CORRECT_PIN_HASH){
      var ov=el('pin-overlay');
      if(ov){ov.style.opacity='0';ov.style.transition='opacity .3s';setTimeout(function(){ov.style.display='none';},300);}
    } else {
      if(err)err.textContent='Incorrect passphrase. Try again.';
      inp.value='';
      inp.focus();
    }
  });
}

var psub=el('pin-submit-btn');
if(psub){
  psub.addEventListener('click',tryPassphrase);
  psub.addEventListener('touchend',function(e){e.preventDefault();tryPassphrase();});
}

var pinp=el('pin-text-input');
if(pinp){
  pinp.addEventListener('keydown',function(e){if(e.key==='Enter'){e.preventDefault();tryPassphrase();}});
}

var pshow=el('pin-show-btn');
if(pshow){
  pshow.addEventListener('click',function(){
    var inp=el('pin-text-input');
    if(!inp)return;
    inp.type=inp.type==='password'?'text':'password';
  });
}

// Auto-skip passphrase if already authenticated
try{
  if(localStorage.getItem('gd_auth')==='1'){
    var ov=el('pin-overlay');
    if(ov)ov.style.display='none';
  }
}catch(e){}

// Sign out button
var so=el('signout-btn');
if(so){
  so.addEventListener('click',function(){
    try{localStorage.removeItem('gd_auth');}catch(e){}
    var ov=el('pin-overlay');
    if(ov){
      ov.style.display='flex';
      ov.style.opacity='1';
      var inp=el('pin-text-input');
      if(inp){inp.value='';}
      var err=el('pin-error');
      if(err)err.textContent='';
    }
    renderTeams();showScr('teams');
  });
}
